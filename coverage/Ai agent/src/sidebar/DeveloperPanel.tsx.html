import { Activity, AlertTriangle, CheckCircle2, RefreshCw, Target, Shield, TrendingUp } from "lucide-react";
import type { AgentState } from "../shared/types/agent";
import type { AgentMetricRecord } from "../debug/AgentMetrics";
import type { ExecutionLogEntry } from "../debug/ExecutionLogger";
import type { HealthCheckResult } from "../ai/healthCheck";
import type { LongTermMemory } from "../types/Memory";

interface DeveloperPanelProps {
  agentState: AgentState | null;
  logs: ExecutionLogEntry[];
  metrics: AgentMetricRecord[];
  memory: LongTermMemory | null;
  healthChecks: HealthCheckResult[];
}

export const DeveloperPanel = ({ agentState, logs, metrics, memory, healthChecks }: DeveloperPanelProps) => (
  <section className="border-b border-zinc-850 bg-zinc-950 p-3.5 text-[10px] text-zinc-400 space-y-3 font-mono">
    <div className="flex items-center gap-1.5 font-bold uppercase tracking-wider text-zinc-500 font-mono">
      <Activity size={12} className="text-[#ff6b35]" />
      Developer Operations Console
    </div>

    <div className="grid grid-cols-1 gap-2.5">
      {/* Agent status */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-350 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5 flex justify-between items-center">
          <span>Agent Diagnostics</span>
          <span className="h-1.5 w-1.5 rounded-full bg-emerald-500 animate-pulse" />
        </div>
        <div className="grid grid-cols-2 gap-1 text-[10px] text-zinc-450">
          <span>Agent: <strong className="text-zinc-300">{agentState?.currentAgent ?? "Unknown"}</strong></span>
          <span>State: <strong className="text-zinc-300">{agentState?.machineState ?? "IDLE"}</strong></span>
          <span>Plan steps: <strong className="text-zinc-300">{agentState?.steps.length ?? 0}</strong></span>
          <span>Errors: <strong className="text-rose-450">{agentState?.errors.length ?? 0}</strong></span>
        </div>
      </div>

      {/* Cognitive Reasoning */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-[#ff6b35] uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          Cognitive Reasoning Output
        </div>
        <div className="text-[10px] text-zinc-400 space-y-1.5">
          <div className="leading-relaxed"><b>Reasoning:</b> {agentState?.reasoning || "—"}</div>
          <div className="flex items-center justify-between text-[9px] border-t border-zinc-900 pt-1.5 mt-1.5">
            <span>Tool: <code className="bg-zinc-950 px-1 py-0.5 rounded text-[8px] border border-zinc-850 text-white font-mono">{agentState?.selectedTool || "—"}</code></span>
            <span>Confidence: <strong className="text-[#ff6b35]">{agentState?.confidence !== undefined ? `${Math.round(agentState.confidence * 100)}%` : "—"}</strong></span>
          </div>
        </div>
      </div>

      {/* Decision Log History */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          Decision History Log
        </div>
        <div className="max-h-24 overflow-y-auto space-y-1.5 text-[9px] text-zinc-450 pr-1 custom-scrollbar">
          {agentState?.decisionHistory?.map((entry, idx) => (
            <div key={idx} className="border-b border-zinc-900/60 pb-1.5 mb-1.5 last:border-0 last:mb-0">
              <div className="flex justify-between font-bold text-zinc-400">
                <span className="text-zinc-300">{entry.action}</span>
                <span className="text-[#ff6b35]">{(entry.confidence * 100).toFixed(0)}%</span>
              </div>
              <p className="italic text-zinc-500 mt-0.5 leading-snug">{entry.reasoning}</p>
              <div className="text-[8px] text-zinc-650 mt-0.5 truncate">Result: {entry.observation}</div>
            </div>
          ))}
          {(!agentState?.decisionHistory || agentState.decisionHistory.length === 0) && (
            <span className="text-zinc-500 italic">No decisions logged yet.</span>
          )}
        </div>
      </div>

      {/* Reflection & Replanning Dashboard */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="flex items-center gap-1 font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          <TrendingUp size={10} className="text-[#ff6b35]" />
          Reflection & Replanning
        </div>
        <div className="grid grid-cols-2 gap-1 text-[10px] text-zinc-450">
          <span className="truncate">Goal: <span className="font-semibold text-zinc-300">{agentState?.goalProgress?.goal ?? agentState?.goal ?? "—"}</span></span>
          <span>Progress: <span className="font-bold text-emerald-450">{agentState?.goalProgress?.completionPercentage ?? 0}%</span></span>
          <span>Failures: <span className="font-medium text-rose-455">{agentState?.failureCount ?? 0}</span></span>
          <span>Replans: <span className="font-medium text-[#ff6b35]">{agentState?.replanCount ?? 0}</span></span>
          <span>Recoveries: <span className="font-medium text-blue-400">{agentState?.recoveryCount ?? 0}</span></span>
          <span>Blocked: <span className={`font-medium ${agentState?.goalProgress?.isBlocked ? "text-rose-500" : "text-emerald-500"}`}>{agentState?.goalProgress?.isBlocked ? "Yes" : "No"}</span></span>
        </div>

        {/* Subgoal list */}
        {agentState?.goalProgress?.subGoals && agentState.goalProgress.subGoals.length > 0 && (
          <div className="mt-1.5 max-h-20 overflow-y-auto space-y-0.5 border-t border-zinc-900 pt-1.5">
            {agentState.goalProgress.subGoals.map((sg) => (
              <div key={sg.id} className="flex items-center gap-1.5 text-zinc-500 text-[9px]">
                {sg.status === "completed" && <CheckCircle2 size={9} className="text-emerald-500 shrink-0" />}
                {sg.status === "failed" && <AlertTriangle size={9} className="text-rose-500 shrink-0" />}
                {sg.status === "running" && <RefreshCw size={9} className="text-[#ff6b35] shrink-0 animate-spin" />}
                {sg.status === "pending" && <span className="h-1.5 w-1.5 rounded-full border border-zinc-750 inline-block shrink-0" />}
                <span className="truncate">{sg.description}</span>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* Execution timeline */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          Execution Timeline Logs
        </div>
        <div className="max-h-24 overflow-y-auto space-y-1.5 text-[9px] text-zinc-455 pr-1 custom-scrollbar">
          {logs.slice(0, 8).map((log) => (
            <div key={log.id} className="flex items-start gap-1.5 leading-normal">
              {log.level === "error" ? <AlertTriangle size={10} className="text-rose-500 shrink-0 mt-0.5" /> : <CheckCircle2 size={10} className="text-emerald-500 shrink-0 mt-0.5" />}
              <span className="break-all"><strong className="text-zinc-300">{log.action ?? "system"}</strong>: {log.message}</span>
            </div>
          ))}
          {logs.length === 0 && <span className="text-zinc-500">No execution logs yet.</span>}
        </div>
      </div>

      {/* Metrics */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          Orchestration Metrics
        </div>
        <div className="grid grid-cols-2 gap-1 text-[10px] text-zinc-450">
          {metrics.slice(0, 6).map((metric) => (
            <span key={`${metric.agent}:${metric.action}`} className="truncate" title={metric.action}>
              {metric.action}: <strong className="text-zinc-300">{metric.runs - metric.failures}/{metric.runs}</strong>
            </span>
          ))}
          {metrics.length === 0 && <span className="text-zinc-500">No metrics yet.</span>}
        </div>
      </div>

      {/* Memory snapshot */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          Memory State Snapshot
        </div>
        <div className="grid grid-cols-2 gap-1 text-[10px] text-zinc-455">
          <span>Saved Jobs: <strong className="text-zinc-300">{memory?.savedJobs.length ?? 0}</strong></span>
          <span>Letters: <strong className="text-zinc-300">{memory?.generatedCoverLetters.length ?? 0}</strong></span>
          <span>Companies: <strong className="text-zinc-300">{memory?.favoriteCompanies.length ?? 0}</strong></span>
          <span>Successful: <strong className="text-zinc-300">{memory?.successfulApplications.length ?? 0}</strong></span>
        </div>
      </div>

      {/* Health */}
      <div className="rounded border border-zinc-900 bg-[#121214] p-2.5 font-mono">
        <div className="font-bold text-zinc-355 uppercase text-[9px] tracking-wide border-b border-zinc-900 pb-1 mb-1.5">
          System Core Diagnostics
        </div>
        <div className="space-y-1.5 text-[9px] text-zinc-455">
          {healthChecks.map((check) => (
            <div key={check.name} className="flex items-center gap-1.5">
              {check.ok ? <CheckCircle2 size={10} className="text-emerald-500" /> : <AlertTriangle size={10} className="text-amber-500" />}
              <span className="truncate">{check.name}: <strong className="text-zinc-350">{check.message}</strong></span>
            </div>
          ))}
          {healthChecks.length === 0 && <span className="text-zinc-500">Diagnostics have not run yet.</span>}
        </div>
      </div>
    </div>
  </section>
);
